cmake_minimum_required(VERSION 3.22)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	project(SyncTERMControllers LANGUAGES C)
	include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/SynctermCStandard.cmake")
	include(CTest)
endif()
foreach(component base/time base/md5 io/filesystem text/utf8)
	get_filename_component(name "${component}" NAME)
	if(NOT TARGET st_${name} AND NOT TARGET syncterm_${name})
		add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../${component}" "${CMAKE_CURRENT_BINARY_DIR}/${name}")
	endif()
endforeach()

add_library(st_controller STATIC controller.c controller_mapping.c controller_descriptor.c controller_native.c controller_xusb.c)
target_include_directories(st_controller PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(st_controller PRIVATE st_time st_md5 st_utf8 syncterm_path syncterm_filesystem)
set_target_properties(st_controller PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(WIN32 OR APPLE)
	include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/SynctermDependencies.cmake")
	syncterm_dependency(NAME HIDAPI PACKAGE "hidapi>=0.14" HEADER hidapi.h LIBRARIES hidapi
		DEFAULT STATIC REQUIRED LINK_REQUIRED PROBE
		"#include <hidapi.h>
int main(void) { unsigned char data[32]; return hid_get_report_descriptor((hid_device *)0, data, sizeof data); }")
	set(ST_CONTROLLER_HIDAPI ON)
	target_link_libraries(st_controller PRIVATE ${ST_DEP_HIDAPI_TARGET})
	if(APPLE)
		set(ST_CONTROLLER_DARWIN_HID ON)
	endif()
else()
	find_package(hidapi 0.14 QUIET CONFIG)
	if(TARGET hidapi::hidraw)
		set(_st_hidapi hidapi::hidraw)
	elseif(TARGET hidapi::netbsd)
		set(_st_hidapi hidapi::netbsd)
	endif()
	find_package(PkgConfig QUIET)
	if(PkgConfig_FOUND AND (NOT CMAKE_CROSSCOMPILING OR DEFINED ENV{PKG_CONFIG_SYSROOT_DIR}) AND NOT _st_hidapi)
		pkg_search_module(ST_CONTROLLER_HID_NATIVE QUIET IMPORTED_TARGET hidapi-hidraw>=0.14 hidapi-netbsd>=0.14)
		if(ST_CONTROLLER_HID_NATIVE_FOUND)
			set(_st_hidapi PkgConfig::ST_CONTROLLER_HID_NATIVE)
		endif()
	endif()
	if(_st_hidapi)
		set(ST_CONTROLLER_HIDAPI ON)
		target_link_libraries(st_controller PRIVATE ${_st_hidapi})
	endif()
endif()
if(ST_CONTROLLER_HIDAPI)
	target_sources(st_controller PRIVATE controller_hid.c)
endif()
set(ST_CONTROLLER_EVDEV OFF)
if(WIN32)
	include(CheckCSourceCompiles)
	include(CMakePushCheckState)
	cmake_push_check_state(RESET)
	set(CMAKE_REQUIRED_LIBRARIES xinput9_1_0)
	check_c_source_compiles("#include <windows.h>
#include <xinput.h>
int main(void) { XINPUT_STATE s; return (int)XInputGetState(0, &s); }" ST_CONTROLLER_XINPUT)
	cmake_pop_check_state()
	if(ST_CONTROLLER_XINPUT)
		target_sources(st_controller PRIVATE controller_xinput.c)
		target_link_libraries(st_controller PRIVATE xinput9_1_0)
	endif()
elseif(APPLE)
	syncterm_dependency(NAME LIBUSB PACKAGE libusb-1.0 HEADER libusb.h LIBRARIES usb-1.0
		DEFAULT STATIC REQUIRED LINK_REQUIRED PROBE
		"#include <libusb.h>
int main(void) { libusb_context *ctx; return libusb_init(&ctx); }")
	set(_st_usb_target ${ST_DEP_LIBUSB_TARGET})
else()
	find_package(PkgConfig QUIET)
	if(PkgConfig_FOUND AND (NOT CMAKE_CROSSCOMPILING OR DEFINED ENV{PKG_CONFIG_SYSROOT_DIR}))
		pkg_check_modules(ST_CONTROLLER_USB QUIET IMPORTED_TARGET libusb-1.0)
		# Kernel-owned XUSB interfaces can expose a USB-parented evdev node.
		pkg_check_modules(ST_CONTROLLER_EVENT_LIBS QUIET IMPORTED_TARGET libudev libevdev)
		if(TARGET PkgConfig::ST_CONTROLLER_EVENT_LIBS)
			include(CheckCSourceCompiles)
			include(CMakePushCheckState)
			cmake_push_check_state(RESET)
			set(CMAKE_REQUIRED_LIBRARIES PkgConfig::ST_CONTROLLER_EVENT_LIBS)
			check_c_source_compiles("#define _POSIX_C_SOURCE 200809L
#include <sys/types.h>
#include <fcntl.h>
#include <libudev.h>
#include <libevdev/libevdev.h>
int main(void) { struct udev *u = udev_new(); struct libevdev *d = 0;
 struct udev_enumerate *e = udev_enumerate_new(u);
 return udev_enumerate_add_match_sysname(e, \"event*\") +
 libevdev_new_from_fd(open(\"/dev/null\", O_NONBLOCK | O_CLOEXEC), &d) + BTN_DPAD_RIGHT; }" ST_CONTROLLER_HAVE_EVDEV_API)
			cmake_pop_check_state()
			if(ST_CONTROLLER_HAVE_EVDEV_API)
				set(ST_CONTROLLER_EVDEV ON)
				target_sources(st_controller PRIVATE controller_evdev.c)
				target_link_libraries(st_controller PRIVATE PkgConfig::ST_CONTROLLER_EVENT_LIBS)
			endif()
		endif()
	endif()
	if(TARGET PkgConfig::ST_CONTROLLER_USB)
		set(_st_usb_target PkgConfig::ST_CONTROLLER_USB)
	endif()
endif()
if(_st_usb_target)
	set(ST_CONTROLLER_LIBUSB ON)
	target_sources(st_controller PRIVATE controller_usb.c)
	target_link_libraries(st_controller PRIVATE ${_st_usb_target})
endif()
configure_file(controller_config.h.in st_controller_config.h)
target_include_directories(st_controller PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(MSVC)
	target_compile_options(st_controller PRIVATE /W4)
	target_compile_definitions(st_controller PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
	target_compile_options(st_controller PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
	add_executable(st_controller_test tests/controller_test.c)
	target_link_libraries(st_controller_test PRIVATE st_controller)
	set_target_properties(st_controller_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	add_test(NAME controller_mapping COMMAND st_controller_test)
endif()
