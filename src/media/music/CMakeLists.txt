include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/SynctermCheckC17Source.cmake")
include(MidiBackends.cmake)
configure_file(music_config.h.in music_config.h @ONLY)
add_library(st_music STATIC ansi_music.c midi_file.c midi_decode.c midi_calibrate.c music.c
	midi_backends.c midi_alsa.c midi_fluidsynth.c midi_adlmidi.c midi_winmm.c midi_apple.c)
target_include_directories(st_music PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
	PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(st_music PUBLIC st_audio PRIVATE syncterm_filesystem st_time syncterm_threads)
target_link_libraries(st_music PRIVATE syncterm_dynamic_library syncterm_string_compat ${_st_midi_deps})
syncterm_target_atomics(st_music)
if(NOT WIN32)
	target_link_libraries(st_music PRIVATE m)
endif()
set_target_properties(st_music PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
	target_compile_options(st_music PRIVATE /W4 $<$<CONFIG:Debug>:/WX>)
else()
	target_compile_options(st_music PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Debug>:-Werror>)
endif()
if(BUILD_TESTING)
	add_executable(st_music_parse_test tests/music_parse_test.c)
	target_link_libraries(st_music_parse_test PRIVATE st_music)
	set_target_properties(st_music_parse_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	add_test(NAME music_parse COMMAND st_music_parse_test)
	add_executable(st_midi_decode_test tests/midi_decode_test.c)
	target_link_libraries(st_midi_decode_test PRIVATE st_music)
	set_target_properties(st_midi_decode_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	add_test(NAME music_decode COMMAND st_midi_decode_test)
	add_executable(st_midi_calibrate_test tests/midi_calibrate_test.c)
	target_link_libraries(st_midi_calibrate_test PRIVATE st_music)
	set_target_properties(st_midi_calibrate_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	add_test(NAME music_calibrate COMMAND st_midi_calibrate_test)
	add_executable(st_music_player_test tests/music_player_test.c)
	target_link_libraries(st_music_player_test PRIVATE st_music syncterm_threads st_time)
	syncterm_target_atomics(st_music_player_test)
	set_target_properties(st_music_player_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	add_test(NAME music_player COMMAND st_music_player_test)
	set_tests_properties(music_player PROPERTIES TIMEOUT 30)
	add_executable(st_midi_synth_test tests/midi_synth_test.c)
	target_link_libraries(st_midi_synth_test PRIVATE st_music)
	set_target_properties(st_midi_synth_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	if(MSVC)
		target_compile_definitions(st_music_parse_test PRIVATE _CRT_SECURE_NO_WARNINGS)
		target_compile_definitions(st_music_player_test PRIVATE _CRT_SECURE_NO_WARNINGS)
	endif()
	if(NOT CMAKE_CROSSCOMPILING)
		find_package(Python3 COMPONENTS Interpreter QUIET)
		if(Python3_Interpreter_FOUND)
			add_test(NAME music_build_modes COMMAND ${Python3_EXECUTABLE}
				${CMAKE_CURRENT_SOURCE_DIR}/tests/midi_build_test.py ${CMAKE_COMMAND}
				${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_C_COMPILER})
			set_tests_properties(music_build_modes PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 240)
		endif()
	endif()
endif()
