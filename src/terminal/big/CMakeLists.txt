add_library(st_big STATIC st_big_number.c st_big_random.c st_big_loop.c st_big_numeric.c
	st_big_profile.c st_big_state.c st_big_control.c st_big_stream.c st_big_draw.c st_big_collect.c st_big_fill.c st_big_flood.c st_big_video.c)
target_include_directories(st_big PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(st_big PUBLIC syncterm_io_error st_graphics_protocol PRIVATE st_time st_fonts st_bgi)
set_target_properties(st_big PROPERTIES
	C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
	target_compile_options(st_big PRIVATE /W4 $<$<CONFIG:Debug>:/WX>)
else()
	target_compile_options(st_big PRIVATE
		-Wall -Wextra -Wpedantic $<$<CONFIG:Debug>:-Werror>)
endif()

if(BUILD_TESTING)
	foreach(part IN ITEMS number random loop numeric state profile control stream draw collect fill bar producer bar3d flood video)
		add_executable(st_big_${part}_test tests/st_big_${part}_test.c)
		target_link_libraries(st_big_${part}_test PRIVATE st_big)
		set_target_properties(st_big_${part}_test PROPERTIES
			C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
		if(MSVC)
			target_compile_options(st_big_${part}_test PRIVATE /W4)
		else()
			target_compile_options(st_big_${part}_test PRIVATE -Wall -Wextra -Wpedantic)
		endif()
		add_test(NAME big_${part} COMMAND st_big_${part}_test)
		set_tests_properties(big_${part} PROPERTIES TIMEOUT 20)
	endforeach()
	target_link_libraries(st_big_number_test PRIVATE st_graphics_protocol)
	target_link_libraries(st_big_loop_test PRIVATE st_graphics_protocol)
	target_link_libraries(st_big_state_test PRIVATE st_bgi)
	target_link_libraries(st_big_profile_test PRIVATE st_fonts st_bgi)
	target_link_libraries(st_big_draw_test PRIVATE st_bgi)
	target_link_libraries(st_big_fill_test PRIVATE st_bgi)
endif()
