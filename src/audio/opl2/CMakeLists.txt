add_library(st_opl2 STATIC st_opl2.c)
target_include_directories(st_opl2 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(st_opl2 PUBLIC syncterm_io_error)
set_target_properties(st_opl2 PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
	target_compile_options(st_opl2 PRIVATE /W4 $<$<CONFIG:Debug>:/WX>)
else()
	target_compile_options(st_opl2 PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Debug>:-Werror>)
endif()
if(BUILD_TESTING)
	add_executable(st_big_opl2_test tests/st_big_opl2_test.c)
	target_link_libraries(st_big_opl2_test PRIVATE st_opl2)
	set_target_properties(st_big_opl2_test PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
	if(MSVC)
		target_compile_options(st_big_opl2_test PRIVATE /W4)
	else()
		target_compile_options(st_big_opl2_test PRIVATE -Wall -Wextra -Wpedantic)
	endif()
	add_test(NAME big_opl2 COMMAND st_big_opl2_test)
	set_tests_properties(big_opl2 PROPERTIES TIMEOUT 20)
endif()
