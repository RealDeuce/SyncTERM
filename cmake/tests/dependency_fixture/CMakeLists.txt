cmake_minimum_required(VERSION 3.22)
project(DependencyFixture C)
if(PROVIDE)
	set(extension c)
	set(implementation "int st_fixture(void) { return 17; }\n")
	if(PROVIDE_CXX)
		enable_language(CXX)
		set(extension cpp)
		set(implementation "#include <string>\nextern \"C\" int st_fixture(void) { return std::stoi(std::to_string(17)); }\n")
	endif()
	if(PROVIDE_RECIPE_API)
		string(APPEND implementation "unsigned int BrotliDecoderVersion(void) { return 1; }\nunsigned int BrotliEncoderVersion(void) { return 1; }\nint cmsGetEncodedCMMversion(void) { return 1; }\n")
	endif()
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.${extension}" "${implementation}")
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.h" "#ifdef __cplusplus\nextern \"C\" {\n#endif\nint st_fixture(void);\n#ifdef __cplusplus\n}\n#endif\n")
	file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/brotli")
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/brotli/decode.h" "unsigned int BrotliDecoderVersion(void);\n")
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/brotli/encode.h" "unsigned int BrotliEncoderVersion(void);\n")
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/lcms2.h" "int cmsGetEncodedCMMversion(void);\n")
	add_library(fixture ${PROVIDE} "${CMAKE_CURRENT_BINARY_DIR}/fixture.${extension}")
	set_target_properties(fixture PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
		CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO
		POSITION_INDEPENDENT_CODE YES WINDOWS_EXPORT_ALL_SYMBOLS YES)
	install(TARGETS fixture ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
	install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fixture.h" DESTINATION include)
	install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/" DESTINATION include)
else()
	include("${PROJECT_MODULES}/SynctermDependencies.cmake")
	set(SYNCTERM_C_STANDARD 17)
	if(NOT DEFINED DEPENDENCY_NAME)
		set(DEPENDENCY_NAME FIXTURE)
	endif()
	set(ST_DEP_${DEPENDENCY_NAME}_SOURCE INSTALLED CACHE STRING "")
	set(cxx)
	if(EXERCISE_CXX)
		enable_language(CXX)
		set(CMAKE_CXX_IMPLICIT_LINK_LIBRARIES "")
		set(cxx CXX)
	endif()
	set(probe "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }")
	if(EXERCISE_RECIPE_PROBE)
		include("${PROJECT_MODULES}/dependencies/Recipes.cmake")
		syncterm_dependency_recipe(${EXERCISE_RECIPE_PROBE})
		set(probe "${recipe_probe}")
	endif()
	set(default RUNTIME)
	if(EXERCISE_NATIVE_SDK)
		# Exercise SDK selection policy with real host-built fixture libraries;
		# this does not emulate a macOS ABI or execute a macOS binary.
		set(APPLE TRUE)
		set(CMAKE_OSX_SYSROOT "${NATIVE_SDK}")
		set(default NATIVE)
		set(ST_PC_FIXTURE_INCLUDE_DIRS /nonexistent-host-package)
		set(ST_PC_FIXTURE_CFLAGS_OTHER -include nonexistent-host-package.h)
	endif()
	syncterm_dependency(NAME ${DEPENDENCY_NAME} HEADER fixture.h LIBRARIES fixture DEFAULT ${default}
		PACKAGE syncterm-nonexistent-fixture REQUIRED ${cxx}
		PROBE "${probe}")
	set(target "${ST_DEP_${DEPENDENCY_NAME}_TARGET}")
	if(NOT TARGET "${target}")
		message(FATAL_ERROR "Fixture dependency was not created")
	endif()
	get_property(mode TARGET ${target} PROPERTY ST_DEP_LINKAGE)
	if(NOT mode STREQUAL EXPECTED)
		message(FATAL_ERROR "Expected ${EXPECTED}, got ${mode}")
	endif()
	if(EXERCISE_NATIVE_SDK AND ST_DEP_FIXTURE_LINKAGE STREQUAL "AUTO")
		get_property(headers TARGET st_dep_fixture PROPERTY ST_DEP_INCLUDE)
		file(REAL_PATH "${headers}" actual_headers)
		file(REAL_PATH "${NATIVE_SDK}/usr/include" expected_headers)
		if(NOT actual_headers STREQUAL expected_headers)
			message(FATAL_ERROR "Native default selected package headers: ${headers}")
		endif()
	endif()
	if(EXERCISE_TRANSITIVE)
		syncterm_dependency(NAME FIXTURE HEADER fixture.h LIBRARIES fixture DEFAULT RUNTIME
			PACKAGE syncterm-nonexistent-fixture REQUIRED LINK_REQUIRED
			PROBE "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }")
		get_property(adapter_mode TARGET st_dep_fixture PROPERTY ST_DEP_LINKAGE)
		get_property(link_mode TARGET ${ST_DEP_FIXTURE_TARGET} PROPERTY ST_DEP_LINKAGE)
		if(NOT adapter_mode STREQUAL "RUNTIME" OR NOT link_mode STREQUAL "SHARED")
			message(FATAL_ERROR "Transitive link changed the runtime adapter's policy")
		endif()
	endif()
	if(EXERCISE_TRANSITIVE OR EXERCISE_CXX)
		file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer.c" "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }\n")
		add_executable(consumer "${CMAKE_CURRENT_BINARY_DIR}/consumer.c")
		set_target_properties(consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
		target_link_libraries(consumer PRIVATE ${ST_DEP_FIXTURE_TARGET})
		if(EXERCISE_CXX)
			file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/consumer-language.txt"
				CONTENT "$<TARGET_PROPERTY:consumer,LINKER_LANGUAGE>")
		endif()
	endif()
endif()
