cmake_minimum_required(VERSION 3.22)
project(DependencyFixture C)
if(PROVIDE)
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.c" "int st_fixture(void) { return 17; }\n")
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.h" "int st_fixture(void);\n")
	add_library(fixture ${PROVIDE} "${CMAKE_CURRENT_BINARY_DIR}/fixture.c")
	set_target_properties(fixture PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
		WINDOWS_EXPORT_ALL_SYMBOLS YES)
	install(TARGETS fixture ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
	install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fixture.h" DESTINATION include)
else()
	include("${PROJECT_MODULES}/SynctermDependencies.cmake")
	set(SYNCTERM_C_STANDARD 17)
	set(ST_DEP_FIXTURE_SOURCE INSTALLED CACHE STRING "")
	set(default RUNTIME)
	if(EXERCISE_NATIVE_SDK)
		# Exercise SDK selection policy with real host-built fixture libraries;
		# this does not emulate a macOS ABI or execute a macOS binary.
		set(APPLE TRUE)
		set(CMAKE_OSX_SYSROOT "${NATIVE_SDK}")
		set(default NATIVE)
		set(ST_PC_FIXTURE_INCLUDE_DIRS /nonexistent-host-package)
		set(ST_PC_FIXTURE_CFLAGS_OTHER -include nonexistent-host-package.h)
	endif()
	syncterm_dependency(NAME FIXTURE HEADER fixture.h LIBRARIES fixture DEFAULT ${default}
		PACKAGE syncterm-nonexistent-fixture REQUIRED
		PROBE "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }")
	if(NOT TARGET st_dep_fixture)
		message(FATAL_ERROR "Fixture dependency was not created")
	endif()
	get_property(mode TARGET st_dep_fixture PROPERTY ST_DEP_LINKAGE)
	if(NOT mode STREQUAL EXPECTED)
		message(FATAL_ERROR "Expected ${EXPECTED}, got ${mode}")
	endif()
	if(EXERCISE_NATIVE_SDK AND ST_DEP_FIXTURE_LINKAGE STREQUAL "AUTO")
		get_property(headers TARGET st_dep_fixture PROPERTY ST_DEP_INCLUDE)
		file(REAL_PATH "${headers}" actual_headers)
		file(REAL_PATH "${NATIVE_SDK}/usr/include" expected_headers)
		if(NOT actual_headers STREQUAL expected_headers)
			message(FATAL_ERROR "Native default selected package headers: ${headers}")
		endif()
	endif()
	if(EXERCISE_TRANSITIVE)
		syncterm_dependency(NAME FIXTURE HEADER fixture.h LIBRARIES fixture DEFAULT RUNTIME
			PACKAGE syncterm-nonexistent-fixture REQUIRED LINK_REQUIRED
			PROBE "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }")
		get_property(adapter_mode TARGET st_dep_fixture PROPERTY ST_DEP_LINKAGE)
		get_property(link_mode TARGET ${ST_DEP_FIXTURE_TARGET} PROPERTY ST_DEP_LINKAGE)
		if(NOT adapter_mode STREQUAL "RUNTIME" OR NOT link_mode STREQUAL "SHARED")
			message(FATAL_ERROR "Transitive link changed the runtime adapter's policy")
		endif()
		file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer.c" "#include <fixture.h>\nint main(void) { return st_fixture() != 17; }\n")
		add_executable(consumer "${CMAKE_CURRENT_BINARY_DIR}/consumer.c")
		set_target_properties(consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
		target_link_libraries(consumer PRIVATE ${ST_DEP_FIXTURE_TARGET})
	endif()
endif()
