# Library-only adapter for pinned LAME, whose upstream build is Autotools.
cmake_minimum_required(VERSION 3.22)
project(SyncTERMLame VERSION 4.0 LANGUAGES C)
include(CheckIncludeFile)
include(CheckTypeSize)
include(CheckCSourceCompiles)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
foreach(header errno.h fcntl.h inttypes.h limits.h stdint.h stdio.h stdlib.h
	strings.h string.h sys/stat.h sys/time.h sys/types.h unistd.h wchar.h)
	string(TOUPPER "${header}" key)
	string(REGEX REPLACE "[^A-Z0-9]" "_" key "${key}")
	check_include_file("${header}" HAVE_${key})
endforeach()
set(CMAKE_EXTRA_INCLUDE_FILES stdint.h)
foreach(type int8_t int16_t int32_t int64_t uint8_t uint16_t uint32_t uint64_t)
	string(TOUPPER "${type}" key)
	check_type_size("${type}" SIZEOF_${key})
	set(HAVE_${key} "${HAVE_SIZEOF_${key}}")
endforeach()
unset(CMAKE_EXTRA_INCLUDE_FILES)
foreach(type "float" "double" "long double" "short" "int" "long" "long long"
	"unsigned short" "unsigned int" "unsigned long" "unsigned long long")
	string(TOUPPER "${type}" key)
	string(REPLACE " " "_" key "${key}")
	check_type_size("${type}" SIZEOF_${key})
endforeach()
set(STDC_HEADERS 1)
set(HAVE_MEMCPY 1)
set(HAVE_STRCHR 1)
set(HAVE_STRTOL 1)
set(HAVE_LONG_DOUBLE 1)
set(USE_FAST_LOG 1)
set(LAME_LIBRARY_BUILD 1)
set(PACKAGE "\"lame\"")
set(PACKAGE_VERSION "\"${PROJECT_VERSION}\"")
set(VERSION "${PACKAGE_VERSION}")
file(READ "${LAME_SOURCE}/config.h.in" template)
string(REGEX REPLACE "#undef ([A-Za-z0-9_]+)" "#cmakedefine \\1 @\\1@" template "${template}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/config.h" CONTENT "${template}" @ONLY)
set(sources VbrTag bitstream encoder fft gain_analysis id3tag lame newmdct presets
	psymodel quantize quantize_pvt reservoir set_get tables takehiro util vbrquantize version mpglib_interface)
list(TRANSFORM sources PREPEND "${LAME_SOURCE}/libmp3lame/")
list(TRANSFORM sources APPEND .c)
add_library(mp3lame ${sources})
set_target_properties(mp3lame PROPERTIES POSITION_INDEPENDENT_CODE YES
	C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO WINDOWS_EXPORT_ALL_SYMBOLS YES)
target_include_directories(mp3lame PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
	"${LAME_SOURCE}" "${LAME_SOURCE}/include" "${LAME_SOURCE}/libmp3lame")
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H=1)
if(MSVC)
	target_compile_definitions(mp3lame PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
check_c_source_compiles("#include <math.h>\nint main(void) { double (*volatile f)(double) = floor; return f(0.5) != 0; }" LAME_MATH_IN_LIBC)
set(private_libs "")
if(NOT LAME_MATH_IN_LIBC)
	target_link_libraries(mp3lame PRIVATE m)
	set(private_libs -lm)
endif()
install(TARGETS mp3lame ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES "${LAME_SOURCE}/include/lame.h" DESTINATION include/lame)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lame.pc"
	"prefix=${CMAKE_INSTALL_PREFIX}\nlibdir=\${prefix}/lib\nincludedir=\${prefix}/include\nName: lame\nDescription: LAME MP3 encoder\nVersion: ${PROJECT_VERSION}\nLibs: -L\${libdir} -lmp3lame\nLibs.private: ${private_libs}\nCflags: -I\${includedir}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lame.pc" DESTINATION lib/pkgconfig)
