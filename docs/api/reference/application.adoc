[[reference-application]]
=== Application API reference

==== Wren VM boundary

The vendored VM's public embedding API is declared and documented upstream in
link:wren_vm/include/wren.h[wren.h].  It is not a SyncTERM-owned compatibility
surface, so this guide does not restate each `wren*` function.  SyncTERM code
normally reaches it only through `st_wren_host`; direct VM consumers link
`st_wren_vm` and follow the upstream slot/handle/foreign-method lifetime rules.

==== SFTP provider contract

The transport-neutral SFTP types live in `sftp_client/st_sftp.h`, shared by
the connection provider and Wren host without a reverse dependency. The
Wren-host C surface and its provider types include:
[source,c]
----

typedef struct st_wren_host st_wren_host_t;
typedef struct st_wren_connected st_wren_connected_t;
typedef struct st_console st_console_t;
typedef struct st_console_display st_console_display_t;

struct st_app_paths;
struct st_audio_runtime;
struct st_profile;
struct st_session_history;

enum st_sftp_operation {
	ST_SFTP_START,
	ST_SFTP_REALPATH,
	ST_SFTP_STAT,
	ST_SFTP_OPENDIR,
	ST_SFTP_READDIR,
	ST_SFTP_CLOSE,
	ST_SFTP_OPEN,
	ST_SFTP_READ,
	ST_SFTP_WRITE,
	ST_SFTP_MKDIR,
	ST_SFTP_RMDIR,
	ST_SFTP_REMOVE,
	ST_SFTP_RENAME,
	ST_SFTP_SET_MTIME,
	ST_SFTP_DESCRIPTIONS,
};

struct st_sftp_request {
	enum st_sftp_operation operation;
	const char *path;
	const char *second_path;
	uint64_t handle;
	uint64_t offset;
	uint64_t modified_time;
	uint32_t flags;
	const void *data;
	size_t size;
};

struct st_sftp_stat {
	uint64_t size;
	int64_t modified_time;
	int64_t accessed_time;
	uint32_t mode;
	uint32_t uid;
	uint32_t gid;
};

struct st_sftp_entry {
	const char *name;
	const char *long_name;
	struct st_sftp_stat stat;
	bool directory;
	bool has_long_description;
	const void *hash;
	size_t hash_size;
};

enum st_sftp_result_kind {
	ST_SFTP_RESULT_STARTED,
	ST_SFTP_RESULT_OK,
	ST_SFTP_RESULT_EOF,
	ST_SFTP_RESULT_STRING,
	ST_SFTP_RESULT_BYTES,
	ST_SFTP_RESULT_HANDLE,
	ST_SFTP_RESULT_STAT,
	ST_SFTP_RESULT_ENTRIES,
	ST_SFTP_RESULT_ERROR,
};

/* Response pointers remain borrowed until the next poll or teardown. */
struct st_sftp_response {
	uint64_t request_id;
	enum st_sftp_result_kind kind;
	const void *data;
	size_t size;
	uint64_t handle;
	struct st_sftp_stat stat;
	const struct st_sftp_entry *entries;
	size_t entry_count;
	int error_code;
	uint32_t server_status;
	const char *message;
	bool transient;
};

struct st_sftp_provider {
	void *context;
	bool (*available)(void *context);
	const char *(*public_directory)(void *context);
	bool (*long_name_extension)(void *context);
	/* submit() must copy every borrowed request pointer before returning and
	 * return an ID unique among requests that have not yet been polled. */
	int (*submit)(void *context, const struct st_sftp_request *request,
		uint64_t *request_id);
	/* Returns one for a response, zero when none is ready, or an error. */
	int (*poll)(void *context, struct st_sftp_response *response,
		uint32_t timeout_ms);
	void (*cancel_all)(void *context);
};

struct st_wren_connected_config {
	st_wren_host_t *host;
	st_console_t *console;
	const struct st_profile *profile;
	const struct st_sftp_provider *sftp;
	st_cterm_t *cterm;
	struct st_connection *connection;
	struct st_term_session *session;
	bool status_enabled;
};

struct st_wren_host_config {
	st_console_display_t *display;
	st_console_t *console;
	/* NULL retains the single-console compatibility path used by tests. */
	st_console_t *picker_console;
	const struct st_app_paths *paths;
	const char *application_title;
	const char *launch_script;
	struct st_session_history *session_history;
	struct st_audio_runtime *audio_runtime;
	/* Borrowed for the host and every connected-context lifetime. */
	st_hyperlink_registry_t *hyperlinks;
	bool safe_mode;
};

/* The console and paths remain borrowed for the host's lifetime. */

	const struct st_wren_host_config *config);
void st_wren_host_destroy(st_wren_host_t *host);

/* Refreshes configured web-list caches with the menu progress overlay. */
int st_wren_host_refresh_web_lists(st_wren_host_t *host);

	struct st_profile *selected, bool *has_selection);
int st_wren_host_prompt(st_wren_host_t *host, const char *title,
	const char *question, const char *initial, size_t max_length,
	char **result);
int st_wren_host_prompt_masked(st_wren_host_t *host, const char *title,
	const char *question, const char *initial, size_t max_length,
	bool masked, char **result);
int st_wren_host_choice(st_wren_host_t *host, const char *title,
	const char *question, const char *const *choices, size_t count);
int st_wren_host_alert(st_wren_host_t *host, const char *title, const char *message);
int st_wren_host_help(st_wren_host_t *host, const char *title, const char *body);
int st_wren_host_disconnected(st_wren_host_t *host, const char *details);
char *st_wren_connected_status_details(st_wren_connected_t *connected);
int st_wren_host_status(st_wren_host_t *host, const char *title, const char *message);

	const char *pem, char **filename);
int st_wren_host_remember_tls_version(st_wren_host_t *host,
	const struct st_profile *profile, int version);
void st_wren_host_apply_tls_version(st_wren_host_t *host, struct st_profile *profile);
int st_wren_host_remember_ssh_key(st_wren_host_t *host,
	const struct st_profile *profile, const uint8_t fingerprint[32],
	bool explicit_update);

/*
 * Creates the same temporary profile as Menu.quickConnect(). An UNKNOWN
 * default uses the BBS-list default; an explicit URL scheme still wins.
 */
int st_wren_host_select_url(st_wren_host_t *host, const char *url,
	int default_connection_type, struct st_profile *selected);

bool st_wren_host_exit_requested(const st_wren_host_t *host);
int st_wren_host_set_screen_mode(st_wren_host_t *host,
	enum st_screen_mode selected);

	enum st_screen_mode *result);
const char *st_wren_host_last_error(const st_wren_host_t *host);

/* Connected Wren lifetime; application connection wiring is deliberately
 * separate from this provider-neutral host. */
int st_wren_connected_create(st_wren_connected_t **result,
	const struct st_wren_connected_config *config);
void st_wren_connected_destroy(st_wren_connected_t *connected);

	uint32_t timeout_ms);

	const char *module, const char *source);

	const st_wren_connected_t *connected);

All pointers inside request/config structs are borrowed unless the surrounding
text explicitly requires the provider to copy them.  Output handles and scalar
outputs are required unless described as optional and are changed only on a
successful zero return.  Host/connected functions return zero or negative
`IO_ERROR_*`; provider poll uses one/zero/negative as described below.

`st_sftp_request` is a borrowed submission record.  The provider's
`submit` callback copies every referenced path/data byte before it returns and
produces an ID unique among requests not yet returned by `poll`.

`st_sftp_response` may contain a borrowed byte string, status message,
stat, or borrowed array of entries; pointers expire at the next `poll` call
or provider destruction, not when the current call returns.
Result kind selects which fields are meaningful.  `transient` tells Wren-side
queue policy whether retry may be useful.

[cols="3m,7"]
|===
| Provider callback | Parameters, result, and effect

| available(context)
| Reports whether the connected transport currently has negotiated, usable
SFTP. Never starts negotiation.

| public_directory(context)
| Returns a provider-owned advertised public directory or NULL.

| long_name_extension(context)
| Returns true when the friendly-name extension was negotiated.

| submit(context, request, request_id)
| Copies the request and queues it; returns zero or `IO_ERROR_*` and writes the
unique ID only on success.

| poll(context, response, timeout_ms)
| Returns one with a response, zero with none ready, or a negative error.

| cancel_all(context)
| Requests cancellation during VM shutdown. Transport close interrupts any
blocked channel setup and joins the worker before freeing its session.
|===

==== Wren host and connected context

`st_wren_host_config` borrows display/menu console, optional picker console,
paths, title/script strings, history, audio runtime, and hyperlink registry for
the host lifetime.  `st_wren_connected_config` borrows host, connected console,
profile, and optional provider for one connected context.

[cols="3m,7"]
|===
| Function | Parameters, result, and effect

| st_wren_host_create(result, config)
| Creates persistent menu VM state and picker support.  Result is unchanged on
failure; caller must keep every configured dependency alive.

| st_wren_host_destroy(host)
| Stops/destroys menu and picker VM state.  Connected contexts must already be
destroyed.

| st_wren_host_refresh_web_lists(host)
| Runs configured web-list cache refresh through the menu progress overlay.

| st_wren_host_run_menu(host, selected, has_selection)
| Pumps the persistent menu until exit or selection.  `selected` must be an
initialized profile; it is replaced only when `has_selection` becomes true.

| st_wren_host_prompt(host, title, question, initial, max_length, result)
| Runs a synchronous prompt in the persistent menu VM over a saved copy of
the connected screen, then restores the screen. `title`, `question`,
`initial`, and `result` are required; `max_length` must not exceed `INT_MAX`.
On success, `*result` is either NULL for cancel or a `malloc`-owned UTF-8
string for the caller to free. Close and resize events are reposted after
the prompt. Returns zero or negative `IO_ERROR_*`; unsupported menu UI
returns `IO_ERROR_NOT_SUPPORTED`.

| st_wren_host_prompt_masked(host, title, question, initial, max_length, masked, result)
| Same contract as prompt; `masked` hides the entered text for credential
questions. Call on the owner/UI thread, outside an active call into this VM.

| st_wren_host_choice(host, title, question, choices, count)
| Shows a synchronous choice dialog, preserving the connected screen and
reposting close/resize events as prompt does. Required strings and labels
are borrowed for the call; `count` is 1 through `INT_MAX`. Returns the
zero-based selected index or a negative error (`IO_ERROR_CANCELLED` on
dismissal). Call on the owner/UI thread, outside an active call into this VM.

| st_wren_host_remember_ssh_key(host, profile, fingerprint, explicit_update)
| Borrows a profile identifying the directory entry by name and a 32-byte
SHA-256 fingerprint. Automatic recording (`explicit_update=false`) updates
only an existing personal entry. Explicit Update also creates a personal
copy of an external/quick-connect profile, opening the list if necessary.
Preserves the entry's current settings rather than overwriting them with a
stale session profile. Safe mode returns zero without writing. Returns zero
or a negative persistence error, including password-required if an unopened
encrypted list cannot be read. Call on the owner/UI thread.

| st_wren_host_alert(host, title, message), st_wren_host_help(host, title, body)
| Required strings are borrowed for a synchronous alert or scrollable help
viewer. Both preserve the screen and enforce the same input barriers and
close/resize forwarding as prompt. Return zero or a negative error. Call on
the owner thread outside an active call into the menu VM.

| st_wren_host_status(host, title, message)
| Draws nonblocking progress, taking a complete console snapshot on the first
update. NULL message clears and restores that snapshot; title may be NULL
only when clearing. Does not claim or discard input. Returns zero or a
negative error; host destruction clears an outstanding overlay.

| st_wren_host_disconnected(host, details)
| Required borrowed details include the snapshotted session status and final
protocol log. After terminal disconnect preparation, displays the themed
bottom-row notice until Enter or process close; F1 opens those details.
Preserves the underlying terminal and restores the blank notice row before
offline history is captured. Uses trusted menu-VM input barriers, does not
invoke connected scripts, and returns zero or a negative UI error.

| st_wren_host_can_trust_tls(host, profile)
| Borrows an entry-identifying profile. True only for an existing personal
entry outside safe mode whose address, port, and trust policy still match.
External, quick-connect, and stale-session entries remain view-only.

| st_wren_host_trust_tls(host, profile, pem, filename)
| All pointers required; profile and PEM are borrowed. Writes a new,
non-overwriting `tls-<sanitized entry name>.pem` beside the configuration,
selects it as the entry's sole certificate trust, disables Web PKI, and resets
the remembered TLS version. Returns zero and a caller-owned filename or a
negative error with output unchanged. Failed writes/entry updates remove the
new file; pre-existing certificate files are never overwritten or removed.
Only available for entries accepted by can_trust_tls.

| st_wren_host_remember_tls_version(host, profile, version)
| Borrows profile and accepts TLS 1.2 or 1.3 enum values. Updates the current
personal entry without overwriting unrelated changes with a stale profile.
Returns zero or a persistence error. Safe mode and nonpersonal entries do not
write. Unsaved observations survive directory reloads until host destruction.

| st_wren_host_apply_tls_version(host, profile)
| Before connecting, reapplies a remembered but unsaved TLS version to an
initialized profile. Both pointers are borrowed. An observation is used only
when entry name, address, port, trust policy, and the original disk version
still match; explicit trust/version edits are not overridden. No file write
occurs.

| st_wren_host_select_url(host, url, default_connection_type, selected)
| Builds the same temporary profile as Wren `Menu.quickConnect`; explicit URL
scheme overrides the supplied default.

| st_wren_host_exit_requested(host)
| Reports persistent menu exit policy.

| st_wren_host_set_screen_mode(host, selected)
| Reconfigures menu and picker consoles to a concrete or configured custom
mode, reloads dimension-compatible fonts, and updates host state.  CURRENT is
not a selectable target.  Returns zero or negative `IO_ERROR_*`.

| st_wren_host_get_screen_mode(host, result)
| Returns the menu's selected screen-mode enum or `IO_ERROR_*`.

| st_wren_host_last_error(host)
| Returns a borrowed diagnostic owned by the host.

| st_wren_connected_create(result, config)
| Creates a fresh per-connection VM, loads connected modules/hooks, and binds
the optional provider-neutral SFTP surface.

| st_wren_connected_destroy(connected)
| Cancels host work and releases VM handles; borrowed dependencies remain
caller-owned.

| st_wren_connected_pump(connected, timeout_ms)
| Polls/resumes queued host async work for up to the relative timeout.

| st_wren_connected_interpret(connected, module, source)
| Compiles/runs source under the named module in this connected VM.

| st_wren_connected_status_details(connected)
| Calls `Status.details` once on the owner thread and copies its returned
string for the disconnect notice. Caller frees the result. NULL context,
absent callable, script failure, non-string return, or allocation failure
returns NULL; the notice still has the default disconnect reason.

| st_wren_connected_shell_closed(connected)
| Delivers the shell-closed lifecycle event exactly through host dispatch.

| st_wren_connected_last_error(connected)
| Returns a borrowed diagnostic owned by the connected context.
|===

The Wren-visible APIs, module availability, fiber resume patterns, hooks, and
safe-mode behavior are documented in link:Wren.adoc[Wren.adoc].

==== Command line and startup policy

[source,c]
----
int st_command_line_parse(struct st_command_line *result, int argc,
    char *const argv[], const char **invalid_argument);
int st_console_startup_configure(struct st_console_startup *result,
    const struct st_settings *settings, const struct st_command_line *command);
bool st_app_should_persist_scaling(bool normal_exit, bool safe_mode,
    enum st_screen_mode startup_mode, enum st_screen_mode current_mode);
int st_terminfo_write(FILE *stream);
----

`result`, settings/command, `argv`, and stream pointers are required.  The
command result and `invalid_argument` borrow `argv` strings; argv must remain
alive while they are used.  Parse/configure return zero or negative
`IO_ERROR_*` and leave complete outputs unchanged on failure.  The persistence
predicate is pure.  Terminfo borrows an open writable stream and does not close
or flush ownership for the caller.

[cols="3m,7"]
|===
| Function | Parameters, result, and effect

| st_command_line_parse(result, argc, argv, invalid_argument)
| Parses help/version/terminfo/run action and every supported option.  Result
borrows strings from `argv`; on invalid input it returns an error and points
`invalid_argument` at the rejected borrowed token.

| st_console_startup_configure(result, settings, command)
| Deterministically merges settings and explicit command-line overrides into
logical `st_console_config`, physical `st_console_display_config`, and initial
fullscreen policy.

| st_app_should_persist_scaling(normal_exit, safe_mode, startup_mode, current_mode)
| Pure policy predicate: scaling persists only for an eligible clean exit and
mode transition outside safe mode.

| st_terminfo_write(stream)
| Writes the complete SyncTERM terminfo source to the caller-owned stream and
reports stream failure.
|===

The `syncterm_app` executable owns the initialization and reverse-order
shutdown of these APIs.  It does not expose another public C entry point beyond
the platform `main`/`st_console_main` integration described in
link:Console.adoc[Console.adoc].
